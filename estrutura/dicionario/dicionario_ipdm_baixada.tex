\section*{Composição dos Índices Sintéticos e Regras Anti-Vazamento}
\begin{itemize}
  \item \textbf{Riqueza:} combina PIB per capita, rendimento médio do emprego formal (RAIS) e consumo residencial de energia elétrica.
  \item \textbf{Longevidade:} combina taxas de mortalidade infantil, perinatal, de 15 a 39 anos e \textbf{de 60 a 69 anos}. Por conter diretamente a resposta $Y$, seu uso como preditor caracteriza \textbf{vazamento de dados (circularidade)}.
  \item \textbf{Escolaridade:} combina atendimento escolar (0--3 anos), proficiência no 5º e 9º ano do Ensino Fundamental e taxa de distorção idade-série no Ensino Médio.
  \item \textbf{IPDM:} média aritmética simples de Riqueza, Longevidade e Escolaridade ($\text{IPDM} = \frac{1}{3}(R + L + E)$). Também excluído como preditor por carregar \texttt{longevidade}.
\end{itemize}

\section*{Observações Operacionais de Leitura em R}
\begin{itemize}
  \item \textbf{Separador e decimal:} o arquivo CSV utiliza ponto e vírgula (\texttt{sep = ";"}) e vírgula decimal (\texttt{dec = ","}), conforme o padrão brasileiro da Fundação Seade.
  \item \textbf{Faltantes:} todas as 10 colunas possuem $0$ valores ausentes (\texttt{colSums(is.na(dados)) == 0}).
  \item \textbf{Dependência (não-i.i.d.):} por se tratar de um painel balanceado de 9 municípios acompanhados ao longo de 6 biênios, observações do mesmo município guardam dependência longitudinal.
\end{itemize}
